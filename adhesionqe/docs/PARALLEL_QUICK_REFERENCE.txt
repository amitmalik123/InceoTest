import pytest

# Parallel-safe (default behavior)
@pytest.mark.parallel
def test_can_run_parallel():
    pass

# Must run alone (not in parallel)
@pytest.mark.serial
def test_modifies_global_state():
    pass

# Group related tests together
@pytest.mark.xdist_group(name="auth")
def test_login():
    pass

@pytest.mark.xdist_group(name="auth")
def test_logout():
    pass  # Runs in same worker as login

    """Get worker information"""
    print(f"Worker: {worker_id}")
    if is_parallel_execution:
        print("Running in parallel")

def test_screenshot(worker_screenshot_path, page):
    """Worker-specific screenshot path"""
    path = worker_screenshot_path("test.png")
    page.screenshot(path=path)

def test_video(worker_video_path):
    """Worker-specific video path"""
    path = worker_video_path("test.webm")

def test_temp(worker_temp_dir):
    """Worker-specific temp directory"""
    file = worker_temp_dir / "data.txt"
    file.write_text("test")

def test_shared(shared_resource_lock):
    """Lock shared resources"""
    with shared_resource_lock("database"):
        # Only one test at a time

┌──────────────────────────────────────────────────────────────────────────┐
│ TROUBLESHOOTING                                                          │
└──────────────────────────────────────────────────────────────────────────┘

Problem: Tests fail in parallel but pass serially
Solution: Mark as serial
  @pytest.mark.serial
  def test_problematic():
      pass

Problem: Uneven load distribution
Solution: Try different strategy
  pytest -n auto --dist loadscope  # Instead of loadfile

Problem: Resource exhaustion
Solution: Limit workers
  pytest -n 4  # Instead of -n auto

Problem: File path conflicts
Solution: Use worker-specific paths
  def test_example(worker_screenshot_path):
      path = worker_screenshot_path("file.png")

Problem: Shared resource conflicts
Solution: Use locks
  def test_example(shared_resource_lock):
      with shared_resource_lock("resource_name"):
          # Access shared resource


┌──────────────────────────────────────────────────────────────────────────┐
│ BEST PRACTICES                                                           │
└──────────────────────────────────────────────────────────────────────────┘

✓ Always test with parallelization: pytest -n auto
✓ Mark serial tests explicitly: @pytest.mark.serial
✓ Use worker-specific paths for files
✓ Group related tests: @pytest.mark.xdist_group(name="group")
✓ Use resource pools for expensive resources
✓ Lock shared resources when needed
✓ Keep tests independent (no shared state)
✓ Clean up resources properly

✗ Don't modify global state without marking @serial
✗ Don't hard-code file paths (use fixtures)
✗ Don't share resources without locks
✗ Don't assume test execution order
