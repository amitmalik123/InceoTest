🔍 JSON SCHEMA VALIDATION
───────────────────────────────────────────────────────────────────────────

Basic:
    from utils.api_contract import JSONSchemaValidator

    validator = JSONSchemaValidator()

    schema = {
        "type": "object",
        "properties": {
            "id": {"type": "integer"},
            "name": {"type": "string"}
        },
        "required": ["id", "name"]
    }

    result = validator.validate_response(response, schema)
    assert result.is_valid

Generate Schema:
    example = {"id": 1, "name": "Test", "active": True}
    schema = validator.create_schema_from_example(example)

🎯 COMPLETE WORKFLOW
───────────────────────────────────────────────────────────────────────────

1. Generate baseline:
    response = requests.get("/api/users/1")
    contract = tester.generate_contract_from_response(
        "/api/users/{id}", "GET", response
    )
    registry.save_contract("get_user_v1", contract)

2. Validate later:
    new_response = requests.get("/api/users/2")
    saved = registry.load_contract("get_user_v1")
    result = tester.test_contract(saved, new_response)
    assert result.is_valid

3. Check changes:
    new = tester.generate_contract_from_response(...)
    changes = tester.compare_contracts(saved, new)
