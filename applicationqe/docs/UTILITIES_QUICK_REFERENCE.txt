🔁 RETRY UTILITIES (utils/retry_utils.py)
───────────────────────────────────────────────────────────────────────────

Decorator:
    @exponential_backoff(max_retries=3, initial_delay=1.0)
    def unstable_operation():
        return api.call()

Retry Until True:
    retry_until_true(
        lambda: page.locator("#status").text == "Done",
        max_attempts=10,
        delay=1.0
    )

Fluent API:
    RetryableAction(lambda: page.click("#btn"))
        .with_retries(3)
        .with_delay(1.0)
        .execute()

⏱️ WAIT UTILITIES (utils/wait_utils.py)
───────────────────────────────────────────────────────────────────────────

Custom Condition:
    wait_for_condition(
        lambda: page.locator(".item").count() == 5,
        timeout=10.0
    )

Text Contains:
    wait_for_text_to_contain(
        page.locator("#status"),
        "Complete",
        timeout=30.0
    )

Wait Context:
    with WaitContext(timeout=10.0) as wait:
        wait.until_visible(page.locator("#modal"))
        wait.until_text_contains(locator, "Success")

💾 STORAGE UTILITIES (utils/storage_utils.py)
───────────────────────────────────────────────────────────────────────────

localStorage:
    storage = LocalStorageManager(page)
    storage.set_item("key", "value")
    storage.set_item("config", {"theme": "dark"})
    value = storage.get_item("key")
    config = storage.get_item_as_json("config")
    storage.save_to_file("backup.json")

Cookies:
    cookies = CookieManager(context)
    cookies.set_cookie("session", "abc123")
    value = cookies.get_cookie_value("session")
    cookies.save_cookies_to_file("cookies.json")

Complete State:
    state = StorageState(page, context)
    state.save_state("session.json")
    state.load_state("session.json")

📥 DOWNLOAD UTILITIES (utils/download_utils.py)

Basic:

CSV Validation:

Hash Validation:

📄 PDF UTILITIES (utils/pdf_utils.py)

Setup:

Basic:
    validator = PDFValidator("document.pdf")
    validator.validate_page_count(min_pages=1)
    text = validator.extract_text()
    validator.assert_text_contains("Invoice")

Advanced:
    validator.validate_metadata(expected_title="Report")

📧 EMAIL UTILITIES (utils/email_utils.py)
───────────────────────────────────────────────────────────────────────────

MailHog Setup:
    docker run -d -p 1025:1025 -p 8025:8025 mailhog/mailhog

MailHog Usage:
    mailhog = MailHogClient("http://localhost:8025")
    mailhog.delete_all_messages()

    email = mailhog.wait_for_email(
        to="user@example.com",
        subject_contains="Welcome",
        timeout=30.0
    )

    mailhog.assert_email_subject_contains(email, "Welcome")
    mailhog.assert_email_body_contains(email, "Thank you")

Mailtrap:
    mailtrap = MailtrapClient(
        api_token="token",
        inbox_id="12345"
    )
    email = mailtrap.wait_for_email(to="test@example.com")

IMAP:
    with IMAPEmailClient("imap.gmail.com", "user@gmail.com", "pass") as imap:
        email_id = imap.wait_for_email(subject_contains="Order")

🎯 EXAMPLE: COMPLETE WORKFLOW
───────────────────────────────────────────────────────────────────────────

# 1. Login with retry
@exponential_backoff(max_retries=3)
def login():
    page.goto("https://app.com/login")
    page.fill("#username", "user")
    page.click("#login")

# 2. Wait for dashboard
with WaitContext(timeout=10.0) as wait:
    wait.until_visible(page.locator("#dashboard"))

# 3. Save state
storage = StorageState(page, context)
storage.save_state("auth.json")

# 4. Download PDF
manager = DownloadManager(page, "downloads")
download = manager.trigger_download(lambda: page.click("#export"))
filepath = manager.save_download(download)

# 5. Validate PDF
pdf = PDFValidator(filepath)
pdf.validate_page_count(min_pages=1)
pdf.assert_text_contains("Report")

# 6. Check email
mailhog = MailHogClient()
email = mailhog.wait_for_email(
    to="user@example.com",
    subject_contains="Report Ready"
)
